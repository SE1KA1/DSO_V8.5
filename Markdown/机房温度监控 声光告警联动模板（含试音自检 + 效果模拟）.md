机房温度监控 声光告警联动模板（含试音自检 + 效果模拟）
遵循工业级设计规范：声光同步触发；LED 采用「绿正常 / 黄预警 / 红告警」通用色标；蜂鸣器推荐 2.3~2.7kHz 有源蜂鸣器（机房噪音下穿透力最强）

二、试音自检专用模板（手动 / 上电触发）
核心设计原则：试音音效与真实告警明显区分，避免运维混淆自检与故障
完整声光自检（测 LED + 蜂鸣器，自动执行 1 次后停止，全程约 1.5 秒）
1.	第 1 段（0~200ms）：绿色 LED 亮 + 蜂鸣器短响 200ms → 停 200ms
2.	第 2 段（400~600ms）：黄色 LED 亮 + 蜂鸣器短响 200ms → 停 200ms
3.	第 3 段（800~1300ms）：红色 LED 亮 + 蜂鸣器长响 500ms → 结束
整体效果模拟：「嘀 嘀 嘀 ——」，同时绿→黄→红依次亮灯，一次性验证所有声光器件
纯蜂鸣器试音（仅测喇叭，LED 不动作）
•	短试音（推荐）：响 200ms → 停 200ms → 响 200ms → 停 200ms → 响 200ms（三声短嘀）
•	长试音：连续长鸣 1 秒（一声长嘀），快速判断音量是否达标

四、可直接复用的代码模板（单片机 / C 语言通用）
// 硬件引脚定义
#define LED_GREEN   PA0
#define LED_YELLOW  PA1
#define LED_RED     PA2
#define BUZZER      PA3

// 告警状态枚举
typedef enum {
    STATUS_NORMAL,    // 正常运行
    STATUS_WARNING,   // 温度预警
    STATUS_ALARM,     // 温度超限
    STATUS_EMERGENCY, // 超温高危
    STATUS_TEST       // 试音自检
} AlarmStatus_t;

// 声光更新函数（放入1ms定时器中断或主循环）
void update_alarm(AlarmStatus_t status, uint32_t sys_tick) {
    // 默认复位：全灭+静音
    digitalWrite(LED_GREEN, LOW);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_RED, LOW);
    digitalWrite(BUZZER, LOW);

    switch(status) {
        case STATUS_NORMAL:
            digitalWrite(LED_GREEN, HIGH);
            break;

        case STATUS_WARNING:
            // 黄灯1秒周期慢闪
            if(sys_tick % 1000 < 500) digitalWrite(LED_YELLOW, HIGH);
            // 蜂鸣器2秒周期间隔鸣响
            if(sys_tick % 2000 < 250) digitalWrite(BUZZER, HIGH);
            break;

        case STATUS_ALARM:
            // 红灯+蜂鸣器同步500ms周期快闪
            if(sys_tick % 500 < 250) {
                digitalWrite(LED_RED, HIGH);
                digitalWrite(BUZZER, HIGH);
            }
            break;

        case STATUS_EMERGENCY:
            // 红灯+蜂鸣器同步200ms周期爆闪
            if(sys_tick % 200 < 100) {
                digitalWrite(LED_RED, HIGH);
                digitalWrite(BUZZER, HIGH);
            }
            break;

        case STATUS_TEST: // 试音自检流程，1500ms完成
            if(sys_tick < 200)        { digitalWrite(LED_GREEN, HIGH); digitalWrite(BUZZER, HIGH); }
            else if(sys_tick < 600)   { digitalWrite(LED_YELLOW, HIGH); digitalWrite(BUZZER, HIGH); }
            else if(sys_tick < 1300)  { digitalWrite(LED_RED, HIGH); digitalWrite(BUZZER, HIGH); }
            break;
    }
}
