二、试音自检专用模板（上电自动 / 手动触发）
核心原则：试音音效与真实告警明显区分，避免混淆自检与真实故障
1. 完整声光自检（同时验证 LED + 蜂鸣器，单次执行不循环，全程约 1.2 秒）
时序流程：
1.	0~200ms：绿色 LED 亮 + 蜂鸣器短响 200ms
2.	300~500ms：黄色 LED 亮 + 蜂鸣器短响 200ms
3.	600~1000ms：红色 LED 亮 + 蜂鸣器长响 400ms
4.	1000ms 后：全灭 + 静音，自检结束
整体效果模拟：「嘀 嘀 嘀 ——」，同时绿→黄→红依次亮灯，一次性验证所有声光器件完好
2. 纯蜂鸣器试音（仅测试喇叭，LED 不动作）
•	短试音（推荐）：响 200ms → 停 200ms → 响 200ms → 停 200ms → 响 200ms（三声短嘀）
•	长试音：连续长鸣 1 秒（一声长嘀），快速判断音量是否达标

四、可直接复用的代码模板（单片机 / C 语言，适配核心板）
// 硬件引脚定义（根据核心板实际引脚修改）
#define LED_GREEN   PA0   // 绿灯
#define LED_YELLOW  PA1   // 黄灯
#define LED_RED     PA2   // 红灯
#define BUZZER      PA3   // 有源蜂鸣器

// 电流检测状态枚举
typedef enum {
    CURRENT_NORMAL,    // 电流正常
    CURRENT_WARNING,   // 电流预警
    CURRENT_ALARM,     // 过流告警
    CURRENT_EMERGENCY, // 严重过流
    CURRENT_TEST       // 试音自检
} CurrentStatus_t;

// 声光更新函数（放入1ms定时器中断或主循环）
void update_current_alarm(CurrentStatus_t status, uint32_t sys_tick) {
    // 默认复位：全灭+静音
    digitalWrite(LED_GREEN, LOW);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_RED, LOW);
    digitalWrite(BUZZER, LOW);

    switch(status) {
        case CURRENT_NORMAL:
            digitalWrite(LED_GREEN, HIGH);
            break;

        case CURRENT_WARNING:
            // 黄灯1秒周期慢闪
            if(sys_tick % 1000 < 500) digitalWrite(LED_YELLOW, HIGH);
            // 蜂鸣器2秒周期间隔鸣响
            if(sys_tick % 2000 < 200) digitalWrite(BUZZER, HIGH);
            break;

        case CURRENT_ALARM:
            // 红灯+蜂鸣器同步500ms周期快闪
            if(sys_tick % 500 < 250) {
                digitalWrite(LED_RED, HIGH);
                digitalWrite(BUZZER, HIGH);
            }
            break;

        case CURRENT_EMERGENCY:
            // 红灯+蜂鸣器同步200ms周期爆闪
            if(sys_tick % 200 < 100) {
                digitalWrite(LED_RED, HIGH);
                digitalWrite(BUZZER, HIGH);
            }
            break;

        case CURRENT_TEST: // 试音自检流程，1200ms完成
            if(sys_tick < 200)        { digitalWrite(LED_GREEN, HIGH); digitalWrite(BUZZER, HIGH); }
            else if(sys_tick < 500)   { digitalWrite(LED_YELLOW, HIGH); digitalWrite(BUZZER, HIGH); }
            else if(sys_tick < 1000)  { digitalWrite(LED_RED, HIGH); digitalWrite(BUZZER, HIGH); }
            break;
    }
}
